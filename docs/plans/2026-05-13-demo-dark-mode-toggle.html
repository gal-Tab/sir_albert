Dark Mode Toggle — Settings Page

User can flip a toggle in Settings → Appearance and the app instantly switches between light and dark themes; the choice persists across sessions.

Architecture: Theme stored in localStorage under app:theme. A ThemeProvider reads it on mount and applies a data-theme attribute to <html>. CSS variables in :root[data-theme="dark"] override the light defaults — no JS recompute on toggle.

Tech stack: Next.js 14 (App Router), React 18, TypeScript 5.4, Vitest 1.5 + React Testing Library

Effort: S (~3h) Flag: theme_toggle_v1 No backend changes

Out of scope: System-preference auto-detect (will follow in v2), per-page theme overrides, theme-aware images.

Step 3 · Implement ThemeProvider
// app/providers/theme-provider.tsx
'use client';
import { createContext, useContext, useEffect, useState } from 'react';

type Theme = 'light' | 'dark';
const Ctx = createContext<{ theme: Theme; setTheme: (t: Theme) => void }>(null!);

export function ThemeProvider({ children }: { children: React.ReactNode }) {
  const [theme, setThemeState] = useState<Theme>('light');

  useEffect(() => {
    const saved = localStorage.getItem('app:theme') as Theme | null;
    if (saved) setThemeState(saved);
  }, []);

  const setTheme = (t: Theme) => {
    setThemeState(t);
    localStorage.setItem('app:theme', t);
    document.documentElement.dataset.theme = t;
  };

  return <Ctx.Provider value={{ theme, setTheme }}>{children}</Ctx.Provider>;
}

export const useTheme = () => useContext(Ctx);

export default function AppearancePage() {
  const { theme, setTheme } = useTheme();
  return (
    <Switch
      role="switch"
      checked={theme === 'dark'}
      onCheckedChange={(v) => setTheme(v ? 'dark' : 'light')}
      label="Dark mode"
    />
  );
}
